cmake_minimum_required(VERSION 3.16)
project(lesson2_card CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
message(STATUS "找到 OpenCV ${OpenCV_VERSION}")
add_executable(lesson2_card main.cpp)
target_include_directories(lesson2_card PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(lesson2_card PRIVATE ${OpenCV_LIBS})
